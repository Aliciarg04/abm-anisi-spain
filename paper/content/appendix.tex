\begin{verbatim}
class Empresa(mesa.Agent):
    """
    Produce bienes y contrata trabajadores.
    Productividad z ~ N(68.0, 5.0) €/hora
    Salario w ~ N(17.5, 2.0) €/hora
    """
    def __init__(self, unique_id, model, 
                 z_media, z_std, w_media, w_std):
        super().__init__(model)
        self.unique_id = unique_id
        
        # Productividad y Salario
        self.z = np.random.normal(z_media, z_std)
        w_provisional = np.random.normal(w_media, w_std)
        # Restricción: w < z (beneficio positivo)
        self.w = max(1.0, min(w_provisional, self.z - 1.0))
        
        self.trabajadores_contratados = 0
        self.demanda_trabajo = 0

    def calcular_demanda(self, G_a_total, num_empresas):
        """
        Demanda de trabajo según ecuación de Anisi:
        L_d = (G_a / n) / (z - w)
        """
        G_a_individual = G_a_total / num_empresas
        if self.z > self.w:
            self.demanda_trabajo = G_a_individual / (self.z - self.w)
        else:
            self.demanda_trabajo = 0

    def step(self):
        pass
\end{verbatim}


\subsection{Clase Trabajador}

\begin{verbatim}
class Trabajador(mesa.Agent):
    """
    Ofrece trabajo de mercado y doméstico.
    z_b ~ N(5.57, 2.0) €/hora (calibrado)
    c ~ N(5.63, 0.05) €/hora (calibrado)
    """
    def __init__(self, unique_id, model, 
                 zb_media, zb_std, c_media, c_std):
        super().__init__(model)
        self.unique_id = unique_id
        self.empleador = None
        self.salario_percibido = 0
        self.horas_mercado = 0
        self.L_s = 0
        
        # Parámetros estocásticos
        self.z_b = max(0.1, np.random.normal(zb_media, zb_std))
        self.c = max(0.1, np.random.normal(c_media, c_std))
        
        # Tecnología de consumo (Anisi)
        self.x = np.random.uniform(15.0, 25.0)
        self.l = np.random.uniform(0.8, 1.2)
        
        self.L_b = 0          # Trabajo doméstico
        self.L_w_total = 0    # Trabajo total
        self.L_wp = 0         # Tiempo potencial

    def step(self):
        # 1. Ingresos de Mercado
        if self.empleador is not None:
            w_real = self.empleador.w
            self.horas_mercado = self.model.admin_publica.j
        else:
            w_real = 0
            self.horas_mercado = 0
        
        renta_bruta = w_real * self.horas_mercado
        C_mercado = renta_bruta * (1 - self.model.admin_publica.t)
        self.salario_percibido = C_mercado
        
        # 2. Consumo deseado
        C_deseado = self.c * self.model.T_i_total
        
        # 3. Oferta nocional (L^s)
        w_ref = w_real if w_real > 0 else self.model.w_promedio
        self.L_s = C_deseado / w_ref if w_ref > 0 else 0
        
        # 4. Trabajo doméstico compensatorio
        deficit = C_deseado - C_mercado
        self.L_b = deficit / self.z_b if deficit > 0 else 0
        
        # 5. Tiempos totales (Anisi)
        self.L_w_total = self.horas_mercado + self.L_b
        tiempo_consumo = self.c * (self.l / self.x)
        self.L_wp = max(0, self.model.T_i_total * (1 - tiempo_consumo))
\end{verbatim}

    Modelo ABM calibrado para España 2024.
    Incluye shocks económicos controlados.
    
    Comportamiento Keynesiano:

    - G_a BAJO → Menos demanda → Menos empleo → MÁS paro
    """
    def __init__(self, 
                 G_a=74092715,    # Calibrado

                 j=1700.0,        # Jornada anual

                 zb_media=5.57, zb_std=2.0,  # Calibrado

\section{Apéndice B: Instrucciones de Uso}

\subsection{Requisitos}

\begin{itemize}
    \item Python 3.10 o superior
    \item Librerías: \texttt{mesa>=3.0}, \texttt{numpy}, \texttt{mesa-viz-tornado}
\end{itemize}

\subsection{Instalación}

\begin{verbatim}
pip install mesa numpy mesa-viz-tornado
\end{verbatim}

\subsection{Ejecución de la Calibración}

\begin{verbatim}
cd Seminario
python calibracion_definitiva.py
\end{verbatim}

Ejecuta 2,000 iteraciones Monte Carlo y muestra los parámetros óptimos.

\subsection{Ejecución de la Visualización Web}

\begin{verbatim}
cd Seminario
python run_web.py
\end{verbatim}

Abrir navegador en: \texttt{http://localhost:8521/}

\subsection{Controles de la Interfaz}

\begin{itemize}
    \item \textbf{Sliders}: Modificar parámetros (G\_a desde 0 hasta 150M €)
    \item \textbf{Reset}: Reiniciar con nuevos parámetros (OBLIGATORIO tras cambiar sliders)
    \item \textbf{Start}: Inicia simulación continua
    \item \textbf{Step}: Avanza un período
\end{itemize}
